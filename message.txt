// CNC Shield V3.0 - Single Joystick X/Y + R3 Toggle for Z-Axis
const int stepPinX = 2;   // X-Axis STEP[cite: 1]
const int dirPinX  = 5;   // X-Axis DIR[cite: 1]
const int stepPinY = 3;   // Y-Axis STEP[cite: 1]
const int dirPinY  = 6;   // Y-Axis DIR[cite: 1]
const int stepPinZ = 4;   // Z-Axis STEP
const int dirPinZ  = 7;   // Z-Axis DIR
const int enPin    = 8;   // Master Enable[cite: 1]

// Standard CNC Shield Analog Header Mapping
const int joyPinX = A0;   // VRX (Left/Right) -> 'Abort' pin
const int joyPinY = A1;   // VRY (Up/Down)    -> 'Hold' pin[cite: 1]
const int joyBtn  = A2;   // SW  (R3 Click)   -> 'Resume' pin

bool zMode = false;              // false = X & Y Mode, true = Z Mode
int lastBtnState = HIGH;
unsigned long lastDebounce = 0;

void setup() {
  pinMode(stepPinX, OUTPUT);
  pinMode(dirPinX, OUTPUT);
  pinMode(stepPinY, OUTPUT);
  pinMode(dirPinY, OUTPUT);
  pinMode(stepPinZ, OUTPUT);
  pinMode(dirPinZ, OUTPUT);
  pinMode(enPin, OUTPUT);
  
  pinMode(joyBtn, INPUT_PULLUP); // R3 button pulls to GND when clicked
  digitalWrite(enPin, LOW);      // Wake up the motors[cite: 1]
  
  Serial.begin(9600);
  Serial.println("Mode: X/Y Axis Control Active");
}

void loop() {
  // --- 1. READ R3 BUTTON TOGGLE ---
  int reading = digitalRead(joyBtn);
  if (reading == LOW && lastBtnState == HIGH && (millis() - lastDebounce > 250)) {
    zMode = !zMode; // Flip mode
    lastDebounce = millis();
    Serial.println(zMode ? "Mode: Z-Axis Active" : "Mode: X/Y Axis Active");
  }
  lastBtnState = reading;

  // --- 2. READ JOYSTICK AXES ---
  int valX = analogRead(joyPinX);
  int valY = analogRead(joyPinY); // Reads 0 to 1023 (Center is ~512)[cite: 1]

  // --- 3. EXECUTE MOTOR MOVEMENT ---
  if (zMode) {
    // Z-MODE: Up/Down moves Z-Axis
    if (valY > 550) {
      digitalWrite(dirPinZ, HIGH);
      int speedDelay = map(valY, 550, 1023, 4000, 500);
      doStep(stepPinZ, speedDelay);
    } 
    else if (valY < 470) {
      digitalWrite(dirPinZ, LOW);
      int speedDelay = map(valY, 470, 0, 4000, 500);
      doStep(stepPinZ, speedDelay);
    }
  } 
  else {
    // X/Y MODE: Dominant axis lockout prevents diagonal/simultaneous movement
    int distX = abs(valX - 512);
    int distY = abs(valY - 512);

    if (distY >= distX) {
      // Up/Down moves Y-Axis (motor 2)
      if (valY > 550) {
        digitalWrite(dirPinY, LOW); // Y-axis inverted[cite: 1]
        int speedDelay = map(valY, 550, 1023, 4000, 500);
        doStep(stepPinY, speedDelay);
      } 
      else if (valY < 470) {
        digitalWrite(dirPinY, HIGH); // Y-axis inverted[cite: 1]
        int speedDelay = map(valY, 470, 0, 4000, 500);
        doStep(stepPinY, speedDelay);
      }
    } 
    else {
      // Left/Right moves X-Axis (motor 1)
      if (valX > 550) {
        digitalWrite(dirPinX, HIGH);
        int speedDelay = map(valX, 550, 1023, 4000, 500);
        doStep(stepPinX, speedDelay);
      } 
      else if (valX < 470) {
        digitalWrite(dirPinX, LOW);
        int speedDelay = map(valX, 470, 0, 4000, 500);
        doStep(stepPinX, speedDelay);
      }
    }
  }
}

void doStep(int stepPin, int usDelay) {
  digitalWrite(stepPin, HIGH);
  delayMicroseconds(usDelay);
  digitalWrite(stepPin, LOW);
  delayMicroseconds(usDelay);
}
